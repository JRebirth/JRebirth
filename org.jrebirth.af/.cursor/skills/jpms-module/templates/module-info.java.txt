/**
 * ${module} JRebirth module.
 *
 * Easiest path: "open module" opens every package for JRebirth/JavaFX reflection,
 * so you never write individual "opens" lines. Switch to a plain "module" plus
 * explicit "opens ... to org.jrebirth.af.core" only when you need strict encapsulation.
 */
open module ${module} {

    // ── JRebirth ──────────────────────────────────────────────────────────────
    requires org.jrebirth.af.api;
    requires org.jrebirth.af.core;
    // requires org.jrebirth.af.component;
    // requires org.jrebirth.af.preloader;
    // requires org.jrebirth.af.iconfontbridge.materialdesignicons;

    // ── JavaFX (add only what the code uses) ───────────────────────────────────
    // requires javafx.controls;   // usually transitive via core
    // requires javafx.fxml;       // DefaultFXMLModel controllers
    // requires javafx.web;        // WebView / HTMLEditor

    // ── JDK / third-party ──────────────────────────────────────────────────────
    // requires java.net.http;
    // requires com.fasterxml.jackson.databind;

    // ── Exports (public API of this module) ────────────────────────────────────
    exports ${pkg};
    // exports ${pkg}.service;

    // ── JRebirth module entry point ────────────────────────────────────────────
    // ${Name}ModuleStarter is generated by the JRebirth annotation processor.
    provides org.jrebirth.af.api.module.ModuleStarter with ${pkg}.${Name}ModuleStarter;

    // ── ServiceLoader SPI (only for real java.util.ServiceLoader services) ─────
    // uses some.spi.Provider;
    // provides some.spi.Provider with ${pkg}.MyProvider;
}
